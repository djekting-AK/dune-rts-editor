    24→    24→// ---------- Deterministic hash for texture variation ----------
    25→    25→function hash(x: number, y: number, seed = 0): number {
    26→    26→  let h = (x * 374761393 + y * 668265263 + seed * 2147483647) | 0
    27→    27→  h = (h ^ (h >>> 13)) * 1274126177
    28→    28→  return ((h ^ (h >>> 16)) >>> 0) / 4294967295
    29→    29→}
    30→    30→function mulberry(seed: number) {
    31→    31→  let a = seed | 0
    32→    32→  return () => {
    33→    33→    a = (a + 0x6D2B79F5) | 0
    34→    34→    let t = a
    35→    35→    t = Math.imul(t ^ (t >>> 15), t | 1)
    36→    36→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    37→    37→    return ((t ^ (t >>> 14)) >>> 0) / 429496296
    38→    38→  }
    39→    39→}

    58→    58→// 2D value noise — smooth, continuous across tile boundaries
    59→    59→function valueNoise(x: number, y: number): number {
    60→    60→  const xi = Math.floor(x), yi = Math.floor(y)
    61→    61→  const xf = x - xi, yf = y - yi
    62→    62→  const fade = (t: number) => t * t * (3 - 2 * t)
    63→    63→  const u = fade(xf), v = fade(yf)
    64→    64→  const a = hash(xi, yi)
    65→    65→  const b = hash(xi + 1, yi)
    66→    66→  const c = hash(xi, yi + 1)
    67→    67→  const d = hash(xi + 1, yi + 1)
    68→    68→  return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v
    69→    69→}
    70→    70→
    71→    71→// Multi-octave noise for richer texture
    72→    72→function fbm(x: number, y: number, octaves = 3): number {
    73→    73→  let sum = 0, amp = 0.5, freq = 1
    74→    74→  for (let i = 0; i < octaves; i++) {
    75→    75→    sum += valueNoise(x * freq, y * freq) * amp
    76→    76→    amp *= 0.5; freq *= 2
    77→    77→  }
    78→    78→  return sum
    79→    79→}

    86→    86→function getSandBase(mapW: number, mapH: number): HTMLCanvasElement {
    87→    87→  const key = `${mapW}x${mapH}`
    88→    88→  let c = sandBaseCache.get(key)
    89→    89→  if (c) return c
    90→    90→  const scale = 2 // render at 1/2 res, upscale smooth
    91→    91→  const w = Math.ceil((mapW * TILE_SIZE) / scale)
    92→    92→  const h = Math.ceil((mapH * TILE_SIZE) / scale)
    93→    93→  c = document.createElement('canvas')
    94→    94→  c.width = w; c.height = h
    95→    95→  const ctx = c.getContext('2d')!
    96→    96→  const img = ctx.createImageData(w, h)
    97→    97→  const data = img.data
    98→    98→  for (let y = 0; y < h; y++) {
    99→    99→    for (let x = 0; x < w; x++) {
   100→   100→      // world-space coordinates (in tile units)
   101→   101→      const wx = (x * scale) / TILE_SIZE
   102→   102→      const wy = (y * scale) / TILE_SIZE
   103→   103→      // multi-octave noise for natural sand variation
   104→   104→      const n = fbm(wx * 0.35, wy * 0.35, 4) // 0..~0.9
   105→   105→      const n2 = fbm(wx * 1.2 + 100, wy * 1.2 + 100, 2) // fine detail
   106→   106→      const t = Math.max(0, Math.min(1, n / 0.7))
   107→   107→      const t2 = Math.max(0, Math.min(1, n2 / 0.6))
   108→   108→      // warm sand palette: dark amber → golden
   109→   109→      const r = Math.round(195 + t * 45 + (t2 - 0.5) * 12)
   110→   110→      const g = Math.round(145 + t * 38 + (t2 - 0.5) * 10)
   111→   111→      const b = Math.round(68 + t * 27 + (t2 - 0.5) * 6)
   112→   112→      const i = (y * w + x) * 4
   113→   113→      data[i] = r; data[i+1] = g; data[i+2] = b; data[i+3] = 255
   114→   114→    }
   115→   115→  }
   116→   116→  ctx.putImageData(img, 0, 0)
   117→   117→  sandBaseCache.set(key, c)
   118→   118→  return c
   119→   119→}

   131→   131→function getTerrainBase(terrain: number[], mapW: number, mapH: number, version: number): HTMLCanvasElement {
   132→   132→  const key = `${mapW}x${mapH}-v${version}`
   133→   133→  let c = terrainBaseCache.get(key)
   134→   134→  if (c) return c
   135→   135→  const scale = 2
   136→   136→  const w = Math.ceil((mapW * TILE_SIZE) / scale)
   137→   137→  const h = Math.ceil((mapH * TILE_SIZE) / scale)
   138→   138→  c = document.createElement('canvas')
   139→   139→  c.width = w; c.height = h
   140→   140→  const ctx = c.getContext('2d')!
   141→   141→  const img = ctx.createImageData(w, h)
   142→   142→  const data = img.data
   143→   143→
   144→   144→  // color function for each terrain type at world coords (continuous noise)
   145→   145→  const typeColor = (t: number, wx: number, wy: number): [number, number, number] => {
   146→   146→    if (t === 0) return [8, 6, 4]
   147→   147→    if (t === 1 || t === 2 || t === 5 || t === 6) {
   148→   148→      const n = fbm(wx * 0.35, wy * 0.35, 4)
   149→   149→      const n2 = fbm(wx * 1.2 + 100, wy * 1.2 + 100, 2)
   150→   150→      const tn = Math.max(0, Math.min(1, n / 0.7))
   151→   151→      const tn2 = Math.max(0, Math.min(1, n2 / 0.6))
   152→   152→      let r = 195 + tn * 45 + (tn2 - 0.5) * 12
   153→   153→      let g = 145 + tn * 38 + (tn2 - 0.5) * 10
   154→   154→      let b = 68 + tn * 27 + (tn2 - 0.5) * 6
   155→   155→      if (t === 5) { r = r * 0.7 + 232 * 0.3; g = g * 0.7 + 93 * 0.3; b = b * 0.7 + 47 * 0.3 }
   156→   156→      else if (t === 6) { r = r * 0.6 + 200 * 0.4; g = g * 0.6 + 60 * 0.4; b = b * 0.6 + 30 * 0.4 }
   157→   157→      if (t === 2) { r *= 0.88; g *= 0.86; b *= 0.82 }
   158→   158→      return [r, g, b]
   159→   159→    }
   160→   160→    if (t === 3) {
   161→   161→      const n = fbm(wx * 0.3, wy * 0.3, 4)
   162→   162→      const n2 = fbm(wx * 1.5 + 50, wy * 1.5 + 50, 2)
   163→   163→      const tn = n / 0.7
   164→   164→      const tn2 = (n2 - 0.5) * 20
   165→   165→      return [118 + tn * 32 + tn2, 113 + tn * 30 + tn2, 103 + tn * 26 + tn2]
   166→   166→    }
   167→   167→    if (t === 4) {
   168→   168→      const n = fbm(wx * 0.3, wy * 0.3, 4)
   169→   169→      return [82 + n * 25, 75 + n * 22, 65 + n * 18]
   170→   170→    }
   171→   171→    if (t === 7) {
   172→   172→      const n = fbm(wx * 0.4, wy * 0.4, 3)
   173→   173→      return [40 + n * 30, 120 + n * 40, 170 + n * 40]
   174→   174→    }
   175→   175→    return [8, 6, 4]
   176→   176→  }
   177→   177→
   178→   178→  const typeAt = (wx: number, wy: number): number => {
   179→   179→    const tx = Math.floor(wx), ty = Math.floor(wy)
   180→   180→    return (tx >= 0 && ty >= 0 && tx < mapW && ty < mapH) ? terrain[ty * mapW + tx] : 0
   181→   181→  }
   182→   182→
   183→   183→  for (let y = 0; y < h; y++) {
   184→   184→    for (let x = 0; x < w; x++) {
   185→   185→      const wx = (x * scale) / TILE_SIZE
   186→   186→      const wy = (y * scale) / TILE_SIZE
   187→   187→      const i = (y * w + x) * 4
   188→   188→      // sample terrain type at this pixel + 4 neighbors for boundary blending
   189→   189→      const t0 = typeAt(wx, wy)
   190→   190→      const offsets = [[0.35, 0], [-0.35, 0], [0, 0.35], [0, -0.35]]
   191→   191→      let r = 0, g = 0, b = 0, total = 0
   192→   192→      // center weight = 1, each neighbor weight = 0.25 (only if different type)
   193→   193→      const [cr, cg, cb] = typeColor(t0, wx, wy)
   194→   194→      r += cr; g += cg; b += cb; total += 1
   195→   195→      for (const [dx, dy] of offsets) {
   196→   196→        const tn = typeAt(wx + dx, wy + dy)
   197→   197→        if (tn !== t0) {
   198→   198→          const [nr, ng, nb] = typeColor(tn, wx + dx, wy + dy)
   199→   199→          r += nr * 0.3; g += ng * 0.3; b += nb * 0.3; total += 0.3
   200→   200→        }
   201→   201→      }
   202→   202→      r /= total; g /= total; b /= total
   203→   203→      data[i] = Math.max(0, Math.min(255, Math.round(r)))
   204→   204→      data[i+1] = Math.max(0, Math.min(255, Math.round(g)))
   205→   205→      data[i+2] = Math.max(0, Math.min(255, Math.round(b)))
   206→   206→      data[i+3] = 255
   207→   207→    }
   208→   208→  }
   209→   209→  ctx.putImageData(img, 0, 0)
   210→   210→  terrainBaseCache.set(key, c)
   211→   211→  for (const k of terrainBaseCache.keys()) {
   212→   212→    if (k !== key) terrainBaseCache.delete(k)
   213→   213→  }
   214→   214→  return c
   215→   215→}

   217→   217→// Render one seamless terrain tile at world position (gx, gy)
   218→   218→// NOTE: sand base is pre-rendered for the whole map — tiles only draw
   219→   219→// features ON TOP of it (dunes ripples, spice glow, rock, water, etc.)
   220→   220→function drawSeamlessTile(
   221→   221→  ctx: CanvasRenderingContext2D,
   222→   222→  tileId: number,
   223→   223→  gx: number, gy: number,
   224→   224→  animPhase: number,
   225→   225→) {
   226→   226→  const s = TILE_SIZE
   227→   227→  const ox = gx * s, oy = gy * s
   228→   228→  ctx.save()
   229→   229→  ctx.translate(ox, oy)
   230→   230→
   231→   231→  if (tileId === 0) {
   232→   232→    ctx.fillStyle = '#080604'
   233→   233→    ctx.fillRect(0, 0, s, s)
   234→   234→  } else if (tileId === 1) {
   235→   235→    // Sand — nothing to draw, sand base shows through (seamless)
   236→   236→    // optional: very faint grain
   237→   237→    const rng = mulberry((gx * 73856093) ^ (gy * 19349663))
   238→   238→    ctx.fillStyle = 'rgba(120,70,20,0.06)'
   239→   239→    for (let i = 0; i < 4; i++) ctx.fillRect(rng() * s, rng() * s, 1, 1)
   240→   240→  } else if (tileId === 2) {
   241→   241→    // Dunes — sand base + ripple lines (continuous wave)
   242→   242→    const rng = mulberry((gx * 73856093) ^ (gy * 19349663))
   243→   243→    ctx.fillStyle = 'rgba(120,70,20,0.06)'
   244→   244→    for (let i = 0; i < 4; i++) ctx.fillRect(rng() * s, rng() * s, 1, 1)
   245→   245→    // dune ripple crests — continuous across tiles via world coords
   246→   246→    ctx.strokeStyle = 'rgba(120,75,20,0.4)'
   247→   247→    ctx.lineWidth = 1.5
   248→   248→    for (let i = 0; i < 3; i++) {
   249→   249→      const yo = i * (s / 3) + 5
   250→   250→      ctx.beginPath()
   251→   251→      ctx.moveTo(-1, yo)
   252→   252→      for (let x = 0; x <= s; x += 3) {
   253→   253→        const wy = yo + Math.sin((gx * s + x) * 0.13 + gy * 0.6) * 3
   254→   254→        ctx.lineTo(x, wy)
   255→   255→      }
   256→   256→      ctx.stroke()
   257→   257→    }
   258→   258→    ctx.strokeStyle = 'rgba(255,225,150,0.32)'
   259→   259→    ctx.lineWidth = 1
   260→   260→    for (let i = 0; i < 3; i++) {
   261→   261→      const yo = i * (s / 3) + 3.5
   262→   262→      ctx.beginPath()
   263→   263→      ctx.moveTo(-1, yo)
   264→   264→      for (let x = 0; x <= s; x += 3) {
   265→   265→        const wy = yo + Math.sin((gx * s + x) * 0.13 + gy * 0.6) * 3
   266→   266→        ctx.lineTo(x, wy)
   267→   267→      }
   268→   268→      ctx.stroke()
   269→   269→    }
   270→   270→  } else if (tileId === 3) {
   271→   271→    // Rock — opaque rocky ground
   272→   272→    const n = fbm(gx * 0.2, gy * 0.2, 3)
   273→   273→    const t = n / 0.7
   274→   274→    const r = Math.round(118 + t * 32)
   275→   275→    const g = Math.round(113 + t * 30)
   276→   276→    const b = Math.round(103 + t * 26)
   277→   277→    ctx.fillStyle = `rgb(${r},${g},${b})`
   278→   278→    ctx.fillRect(0, 0, s, s)
   279→   279→    const rng = mulberry((gx * 12345) ^ (gy * 54321))
   280→   280→    ctx.fillStyle = 'rgba(50,45,35,0.45)'
   281→   281→    for (let i = 0; i < 4; i++) {
   282→   282→      ctx.fillRect(rng() * s, rng() * s, 2 + rng() * 4, 1)
   283→   283→    }
   284→   284→    ctx.fillStyle = 'rgba(180,175,165,0.35)'
   285→   285→    for (let i = 0; i < 5; i++) ctx.fillRect(rng() * s, rng() * s, 1, 1)
   286→   286→  } else if (tileId === 4) {
   287→   287→    // Mountain — rocky base + soft peak + snow
   288→   288→    const n = fbm(gx * 0.2, gy * 0.2, 3)
   289→   289→    ctx.fillStyle = `rgb(${82 + n * 22},${75 + n * 20},${65 + n * 16})`
   290→   290→    ctx.fillRect(0, 0, s, s)
   291→   291→    const peakGrad = ctx.createRadialGradient(s / 2, s * 0.6, 2, s / 2, s * 0.6, s * 0.55)
   292→   292→    peakGrad.addColorStop(0, '#9a8e7e')
   293→   293→    peakGrad.addColorStop(0.5, '#6a5e50')
   294→   294→    peakGrad.addColorStop(1, 'rgba(58,50,40,0)')
   295→   295→    ctx.fillStyle = peakGrad
   296→   296→    ctx.fillRect(0, 0, s, s)
   297→   297→    const snow = ctx.createRadialGradient(s / 2, s * 0.25, 1, s / 2, s * 0.25, s * 0.2)
   298→   298→    snow.addColorStop(0, 'rgba(245,245,250,0.95)')
   299→   299→    snow.addColorStop(1, 'rgba(245,245,250,0)')
   300→   300→    ctx.fillStyle = snow
   301→   301→    ctx.fillRect(0, 0, s, s)
   302→   302→  } else if (tileId === 5 || tileId === 6) {
   303→   303→    // Spice — sand base shows through + glowing crystals (semi-transparent)
   304→   304→    const rich = tileId === 6
   305→   305→    ctx.fillStyle = rich ? 'rgba(200,60,30,0.22)' : 'rgba(232,93,47,0.16)'
   306→   306→    ctx.fillRect(0, 0, s, s)
   307→   307→    const rng = mulberry((gx * 3333) ^ (gy * 7777))
   308→   308→    const count = rich ? 5 : 4
   309→   309→    for (let i = 0; i < count; i++) {
   310→   310→      const cx = 4 + rng() * (s - 8), cy = 4 + rng() * (s - 8)
   311→   311→      const rad = 4 + rng() * 4
   312→   312→      const glow = ctx.createRadialGradient(cx, cy, 0, cx, cy, rad)
   313→   313→      glow.addColorStop(0, rich ? 'rgba(255,100,50,0.85)' : 'rgba(255,140,70,0.7)')
   314→   314→      glow.addColorStop(0.5, rich ? 'rgba(220,60,30,0.5)' : 'rgba(232,93,47,0.4)')
   315→   315→      glow.addColorStop(1, 'rgba(232,93,47,0)')
   316→   316→      ctx.fillStyle = glow
   317→   317→      ctx.fillRect(cx - rad, cy - rad, rad * 2, rad * 2)
   318→   318→      ctx.fillStyle = rich ? '#ff7050' : '#ff9060'
   319→   319→      ctx.fillRect(cx - 1, cy - 1, 2, 2)
   320→   320→    }
   321→   321→  } else if (tileId === 7) {
   322→   322→    // Water — opaque, animated continuous ripples
   323→   323→    const g = ctx.createLinearGradient(0, 0, 0, s)
   324→   324→    g.addColorStop(0, '#3aa0c8'); g.addColorStop(1, '#1a5e80')
   325→   325→    ctx.fillStyle = g; ctx.fillRect(0, 0, s, s)
   326→   326→    ctx.strokeStyle = 'rgba(150,220,240,0.4)'
   327→   327→    ctx.lineWidth = 1
   328→   328→    const phase = animPhase * 0.6
   329→   329→    for (let i = 0; i < 4; i++) {
   330→   330→      const yo = (i * (s / 4) + phase) % s
   331→   331→      ctx.beginPath()
   332→   332→      ctx.moveTo(-1, yo)
   333→   333→      for (let x = 0; x <= s; x += 3) {
   334→   334→        const wy = yo + Math.sin((gx * s + x) * 0.2 + phase) * 1.2
   335→   335→        ctx.lineTo(x, wy)
   336→   336→      }
   337→   337→      ctx.stroke()
   338→   338→    }
   339→   339→    const rng = mulberry((gx * 999) ^ (gy * 111))
   340→   340→    ctx.fillStyle = 'rgba(220,245,255,0.6)'
   341→   341→    ctx.fillRect((phase * 3 + rng() * s) % s, 3, 2, 1)
   342→   342→  }
   343→   343→
   344→   344→  ctx.restore()
   345→   345→}

   347→   347→// Soft edge blending: feather borders where rock/water/mountain meets sand
   348→   348→function blendEdges(
   349→   349→  ctx: CanvasRenderingContext2D,
   350→   350→  tileId: number,
   351→   351→  gx: number, gy: number,
   352→   352→  terrain: number[], w: number, h: number,
   353→   353→) {
   354→   354→  const s = TILE_SIZE
   355→   355→  const ox = gx * s, oy = gy * s
   356→   356→  if (tileId === 0 || tileId === 1 || tileId === 2) return
   357→   357→  const neighbors = [
   358→   358→    { dx: -1, dy: 0 }, { dx: 1, dy: 0 }, { dx: 0, dy: -1 }, { dx: 0, dy: 1 },
   359→   359→  ]
   360→   360→  for (const nb of neighbors) {
   361→   361→    const nx = gx + nb.dx, ny = gy + nb.dy
   362→   362→    if (nx < 0 || ny < 0 || nx >= w || ny >= h) continue
   363→   363→    const nt = terrain[ny * w + nx]
   364→   364→    if (nt === tileId || nt === 0) continue
   365→   365→    if ((nt === 1 || nt === 2) && (tileId === 3 || tileId === 7 || tileId === 4)) {
   366→   366→      const grad = ctx.createLinearGradient(
   367→   367→        nb.dx === -1 ? ox : nb.dx === 1 ? ox + s : ox,

   383→   383→// ---------- Public: draw full seamless terrain layer ----------
   384→   384→// Blits the continuous terrain base (per-pixel, no squares) then draws
   385→   385→// feature overlays on top (dune ripples, spice glow, water animation, peaks).
   386→   386→export function drawTerrainLayer(
   387→   387→  ctx: CanvasRenderingContext2D,
   388→   388→  terrain: number[], w: number, h: number,
   389→   389→  animPhase: number,
   390→   390→  version = 0,
   391→   391→) {
   392→   392→  // 1. continuous terrain base — ALL types per-pixel noise, truly seamless
   393→   393→  const base = getTerrainBase(terrain, w, h, version)
   394→   394→  ctx.imageSmoothingEnabled = true
   395→   395→  ctx.drawImage(base, 0, 0, w * TILE_SIZE, h * TILE_SIZE)
   396→   396→  // 2. per-tile feature overlays (semi-transparent, sand base shows through)
   397→   397→  for (let y = 0; y < h; y++) {
   398→   398→    for (let x = 0; x < w; x++) {
   399→   399→      const t = terrain[y * w + x]
   400→   400→      if (t === 2) {
   401→   401→        // dunes: ripple crests on top of sand-colored base
   402→   402→        drawDuneRipples(ctx, x, y)
   403→   403→      } else if (t === 5 || t === 6) {
   404→   404→        // spice: glowing crystal clusters
   405→   405→        drawSpiceGlow(ctx, x, y, t === 6)
   406→   406→      } else if (t === 7) {
   407→   407→        // water: animated ripples
   408→   408→        drawWaterAnim(ctx, x, y, animPhase)
   409→   409→      } else if (t === 4) {
   410→   410→        // mountain: peak + snow cap overlay
   411→   411→        drawMountainPeak(ctx, x, y)
   412→   412→      }
   413→   413→    }
   414→   414→  }
   415→   415→}

   417→   417→// Feature overlays — drawn semi-transparently over the seamless base
   418→   418→function drawDuneRipples(ctx: CanvasRenderingContext2D, gx: number, gy: number) {
   419→   419→  const s = TILE_SIZE
   420→   420→  ctx.save()
   421→   421→  ctx.translate(gx * s, gy * s)
   422→   422→  // ripples use world-space Y so they flow continuously across tiles
   423→   423→  ctx.strokeStyle = 'rgba(120,75,20,0.3)'
   424→   424→  ctx.lineWidth = 1.5
   425→   425→  for (let i = 0; i < 4; i++) {
   426→   426→    const yo = ((gy * s + i * 10 + 5) % s)
   427→   427→    ctx.beginPath()
   428→   428→    ctx.moveTo(-1, yo)
   429→   429→    for (let x = 0; x <= s; x += 3) {
   430→   430→      const wy = yo + Math.sin((gx * s + x) * 0.13 + gy * 0.6) * 3
   431→   431→      ctx.lineTo(x, wy)
   432→   432→    }
   433→   433→    ctx.stroke()
   434→   434→  }
   435→   435→  ctx.strokeStyle = 'rgba(255,225,150,0.2)'
   436→   436→  ctx.lineWidth = 1
   437→   437→  for (let i = 0; i < 4; i++) {
   438→   438→    const yo = ((gy * s + i * 10 + 3) % s)
   439→   439→    ctx.beginPath()
   440→   440→    ctx.moveTo(-1, yo)
   441→   441→    for (let x = 0; x <= s; x += 3) {
   442→   442→      const wy = yo + Math.sin((gx * s + x) * 0.13 + gy * 0.6) * 3
   443→   443→      ctx.lineTo(x, wy)
   444→   444→    }
   445→   445→    ctx.stroke()
   446→   446→  }
   447→   447→  ctx.restore()
   448→   448→}

   450→   450→function drawSpiceGlow(ctx: CanvasRenderingContext2D, gx: number, gy: number, rich: boolean) {
   451→   451→  const s = TILE_SIZE
   452→   452→  ctx.save()
   453→   453→  ctx.translate(gx * s, gy * s)
   454→   454→  // uniform spice field overlay — clear orange tint, not scattered dots
   455→   455→  ctx.fillStyle = rich ? 'rgba(200,50,20,0.4)' : 'rgba(232,93,47,0.3)'
   456→   456→  ctx.fillRect(0, 0, s, s)
   457→   457→  // a few crystal clusters (deterministic) — bright spots
   458→   458→  const rng = mulberry((gx * 3333) ^ (gy * 7777))
   459→   459→  const count = rich ? 3 : 2
   460→   460→  for (let i = 0; i < count; i++) {
   461→   461→    const cx = 6 + rng() * (s - 12), cy = 6 + rng() * (s - 12)
   462→   462→    const rad = 5 + rng() * 3
   463→   463→    const glow = ctx.createRadialGradient(cx, cy, 0, cx, cy, rad)
   464→   464→    glow.addColorStop(0, rich ? 'rgba(255,120,60,0.8)' : 'rgba(255,150,80,0.65)')
   465→   465→    glow.addColorStop(1, 'rgba(232,93,47,0)')
   466→   466→    ctx.fillStyle = glow
   467→   467→    ctx.fillRect(cx - rad, cy - rad, rad * 2, rad * 2)
   468→   468→  }
   469→   469→  // sparkle highlights
   470→   470→  ctx.fillStyle = rich ? '#ffb070' : '#ffd090'
   471→   471→  for (let i = 0; i < 3; i++) {
   472→   472→    ctx.fillRect(4 + rng() * (s - 8), 4 + rng() * (s - 8), 1, 1)
   473→   473→  }
   474→   474→  ctx.restore()
   475→   475→}

   477→   477→function drawWaterAnim(ctx: CanvasRenderingContext2D, gx: number, gy: number, animPhase: number) {
   478→   478→  const s = TILE_SIZE

   483→   483→  const phase = animPhase * 0.6

   488→   488→    for (let x = 0; x <= s; x += 3) {
   489→   489→      const wy = yo + Math.sin((gx * s + x) * 0.2 + phase) * 1.2
   490→   490→      ctx.lineTo(x, wy)
   491→   491→    }
   492→   492→    ctx.stroke()
   493→   493→  }
   494→   494→  ctx.restore()

  1037→  1037→export function drawExplosion(ctx: CanvasRenderingContext2D, x: number, y: number, frame: number, maxFrame: number, size: number, color: string) {
  1038→  1038→  const t = frame / maxFrame
