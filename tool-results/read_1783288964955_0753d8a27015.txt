    55→// Generate a varied, visually interesting default map (sand + spice + dunes + rock + water)
    56→function generateDefaultMap(w: number, h: number): number[] {
    57→  const g = new Array(w * h).fill(1)
    58→  const rng = (() => { let s = 12345; return () => { s = (s * 1103515245 + 12345) & 0x7fffffff; return s / 0x7fffffff } })()
    59→  // dune patches
    60→  for (let i = 0; i < 5; i++) {
    61→    const cx = Math.floor(rng() * w), cy = Math.floor(rng() * h), r = 2 + Math.floor(rng() * 3)
    62→    for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) {
    63→      const x = cx + dx, y = cy + dy
    64→      if (x < 0 || y < 0 || x >= w || y >= h) continue
    65→      if (dx*dx + dy*dy <= r*r && rng() > 0.3) g[y*w+x] = 2
    66→    }
    67→  }
    68→  // rock plateaus
    69→  for (let i = 0; i < 4; i++) {
    70→    const cx = Math.floor(rng() * w), cy = Math.floor(rng() * h), r = 1 + Math.floor(rng() * 2)
    71→    for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) {
    72→      const x = cx + dx, y = cy + dy
    73→      if (x < 0 || y < 0 || x >= w || y >= h) continue
    74→      if (dx*dx + dy*dy <= r*r) g[y*w+x] = 3
    75→    }
    76→  }
    77→  // mountains (on rock)
    78→  for (let i = 0; i < 3; i++) {
    79→    const cx = Math.floor(rng() * w), cy = Math.floor(rng() * h)
    80→    if (g[cy*w+cx] === 3) g[cy*w+cx] = 4
    81→  }
    82→  // water lakes
    83→  for (let i = 0; i < 2; i++) {
    84→    const cx = Math.floor(rng() * w), cy = Math.floor(rng() * h), r = 2
    85→    for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) {
    86→      const x = cx + dx, y = cy + dy
    87→      if (x < 0 || y < 0 || x >= w || y >= h) continue
    88→      if (dx*dx + dy*dy <= r*r) g[y*w+x] = 7
    89→    }
    90→  }
    91→  // spice fields — ONLY on sand (tile 1), fewer but larger patches, clearly distinct
    92→  for (let i = 0; i < 5; i++) {
    93→    const cx = Math.floor(rng() * w), cy = Math.floor(rng() * h), r = 2 + Math.floor(rng() * 2)
    94→    for (let dy = -r; dy <= r; dy++) for (let dx = -r; dx <= r; dx++) {
    95→      const x = cx + dx, y = cy + dy
    96→      if (x < 0 || y < 0 || x >= w || y >= h) continue
    97→      const idx = y*w+x
    98→      if (g[idx] === 1 && rng() > 0.3) g[idx] = rng() > 0.45 ? 5 : 6
    99→    }
   100→  }
   101→  // clear corners for bases (atreides bottom-left, harkonnen top-right)
   102→  for (let dy = -2; dy <= 2; dy++) for (let dx = -2; dx <= 2; dx++) {
   103→    if (Math.abs(dx) + Math.abs(dy) <= 3) {
   104→      if (3+dx >= 0 && Math.floor(h/2)+dy >= 0 && 3+dx < w && Math.floor(h/2)+dy < h) g[(Math.floor(h/2)+dy)*w + (3+dx)] = 1
   105→      if (w-4+dx >= 0 && Math.floor(h/2)+dy >= 0 && w-4+dx < w && Math.floor(h/2)+dy < h) g[(Math.floor(h/2)+dy)*w + (w-4+dx)] = 1
   106→    }
   107→  }
   108→  return g
   109→}

   111→function floodFill(grid: number[], w: number, h: number, x: number, y: number, target: number, repl: number): number[] {
   112→  if (target === repl) return grid
   113→  const r = [...grid]; const st = [[x, y]]
   114→  while (st.length) {
   115→    const [cx, cy] = st.pop()!
   116→    if (cx < 0 || cy < 0 || cx >= w || cy >= h) continue
   117→    const i = cy * w + cx
   118→    if (r[i] !== target) continue
   119→    r[i] = repl
   120→    st.push([cx+1,cy],[cx-1,cy],[cx,cy+1],[cx,cy-1])
   121→  }
   122→  return r
   123→}

   489→function GameScreen({ difficulty, terrain, w, h, onExit, isFullscreen, toggleFullscreen }: {
   490→  difficulty: 'easy'|'medium'|'hard'; terrain: number[]; w: number; h: number; onExit: ()=>void
   491→  isFullscreen: boolean; toggleFullscreen: ()=>void
   492→}) {
   493→  const canvasRef = useRef<HTMLCanvasElement>(null)
   494→  const gameRef = useRef<GameState>(createGame(w, h, [...terrain], difficulty))
   495→
   496→  // Center viewport on player's palace at game start
   497→  const centerOnBase = useCallback(() => {
   498→    const wrap = canvasWrapRef.current
   499→    if (!wrap) return
   500→    const s = gameRef.current
   501→    const palace = s.buildings.find(b => b.owner === 'atreides' && b.type === 'palace')
   502→    if (!palace) return
   503→    const z = zoomRef.current
   504→    // palace center in world pixels
   505→    const baseX = (palace.x + palace.w / 2) * TILE_SIZE
   506→    const baseY = (palace.y + palace.h / 2) * TILE_SIZE
   507→    // pan so base center → screen center: pan = screenCenter - baseCenter*zoom
   508→    const panX = wrap.clientWidth / 2 - baseX * z
   509→    const panY = wrap.clientHeight / 2 - baseY * z
   510→    const np = { x: panX, y: panY }
   511→    panRef.current = np
   512→    setPan(np)
   513→  }, [])
   514→
   515→  // Center on base when game starts (after canvas mount)
   516→  useEffect(() => {
   517→    const t = setTimeout(centerOnBase, 100)
   518→    return () => clearTimeout(t)
   519→  }, [centerOnBase])
   520→  // offscreen cache for terrain layer (expensive to redraw every frame)
   521→  const terrainCacheRef = useRef<HTMLCanvasElement | null>(null)
   522→  const terrainCacheVerRef = useRef<number>(-1)
   523→  // offscreen cache for fog of war (updated every few ticks)
   524→  const fogCacheRef = useRef<HTMLCanvasElement | null>(null)
   525→  const fogCacheTickRef = useRef<number>(-1)
   526→  const [, forceRender] = useState(0)
   527→  const [paused, setPaused] = useState(false)
   528→  const [selected, setSelected] = useState<{type:'unit'|'building', id:number} | null>(null)
   529→  const [buildMode, setBuildMode] = useState<BuildingType | null>(null)
   530→  const [hoverCell, setHoverCell] = useState<{x:number,y:number}|null>(null)
   531→  const [showHelp, setShowHelp] = useState(false)
   532→  const [zoom, setZoom] = useState(1.0)
   533→  const [pan, setPan] = useState<{x: number; y: number}>({x: 0, y: 0})
   534→  const [panelOpen, setPanelOpen] = useState(false)
   535→  const canvasWrapRef = useRef<HTMLDivElement>(null)
   536→  const selectedRef = useRef(selected); selectedRef.current = selected
   537→  const buildModeRef = useRef(buildMode); buildModeRef.current = buildMode
   538→  const pausedRef = useRef(paused); pausedRef.current = paused
   539→  const zoomRef = useRef(zoom); zoomRef.current = zoom
   540→  const panRef = useRef(pan); panRef.current = pan
   541→  const hoverCellRef = useRef(hoverCell); hoverCellRef.current = hoverCell
   542→  const panelOpenRef = useRef(panelOpen); panelOpenRef.current = panelOpen
   543→  // mouse middle-button pan
   544→  const mousePanRef = useRef<{x:number;y:number;panning:boolean} | null>(null)
   545→  // Pointer events (unified mouse + touch)
   546→  const pointerStateRef = useRef<{startX:number;startY:number;lastX:number;lastY:number;moved:boolean;isDown:boolean;isTwoFinger:boolean;pinchDist:number}>({startX:0,startY:0,lastX:0,lastY:0,moved:false,isDown:false,isTwoFinger:false,pinchDist:0})
   547→  // touch state: track single-finger drag for pan vs tap
   548→  const touchStateRef = useRef<{startX:number, startY:number, lastX:number, lastY:number, moved:boolean, startTime:number}>({startX:0, startY:0, lastX:0, lastY:0, moved:false, startTime:0})
   549→
   550→  // ---- zoom with mouse wheel ----
   551→  const onWheel = (e: React.WheelEvent) => {
   552→    e.preventDefault()
   553→    setZoom(z => {
   554→      const nz = Math.max(0.7, Math.min(2.5, +(z - Math.sign(e.deltaY) * 0.15).toFixed(2)))
   555→      zoomRef.current = nz
   556→      return nz
   557→    })
   558→  }
   559→  // ---- pinch-to-zoom (two fingers) + pan (one finger drag) ----
   560→  const pinchRef = useRef<{ dist: number } | null>(null)
   561→  const handleTouchStart = (e: React.TouchEvent) => {
   562→    if (e.touches.length === 2) {
   563→      const t0 = e.touches[0], t1 = e.touches[1]
   564→      pinchRef.current = { dist: Math.hypot(t0.clientX - t1.clientX, t0.clientY - t1.clientY) }
   565→      touchStateRef.current.moved = true  // disable tap when pinching
   566→    } else if (e.touches.length === 1) {
   567→      const t = e.touches[0]
   568→      touchStateRef.current = { startX: t.clientX, startY: t.clientY, lastX: t.clientX, lastY: t.clientY, moved: false, startTime: Date.now() }
   569→    }
   570→  }
   571→  const handleTouchMove = (e: React.TouchEvent) => {
   572→    if (e.touches.length === 2) {
   573→      // pinch zoom
   574→      e.preventDefault()
   575→      const t0 = e.touches[0], t1 = e.touches[1]
   576→      const dist = Math.hypot(t0.clientX - t1.clientX, t0.clientY - t1.clientY)
   577→      if (pinchRef.current) {
   578→        const ratio = dist / pinchRef.current.dist
   579→        setZoom(z => {
   580→          const nz = Math.max(0.7, Math.min(2.5, +(z * ratio).toFixed(2)))
   581→          zoomRef.current = nz
   582→          return nz
   583→        })
   584→      }
   585→      pinchRef.current = { dist }
   586→    } else if (e.touches.length === 1) {
   587→      const t = e.touches[0]
   588→      const ts = touchStateRef.current
   589→      const dx = t.clientX - ts.lastX
   590→      const dy = t.clientY - ts.lastY
   591→      const totalDx = t.clientX - ts.startX
   592→      const totalDy = t.clientY - ts.startY
   593→      // if moved more than 8px, it's a drag (pan), not a tap
   594→      if (Math.hypot(totalDx, totalDy) > 8) {
   595→        ts.moved = true
   596→        e.preventDefault()
   597→        setPan(p => {
   598→          const np = { x: p.x + dx, y: p.y + dy }
   599→          panRef.current = np
   600→          return np
   601→        })
   602→      }
   603→      ts.lastX = t.clientX
   604→      ts.lastY = t.clientY
   605→    }
   606→  }
   607→  const handleTouchEnd = (e: React.TouchEvent) => {
   608→    pinchRef.current = null
   609→    const ts = touchStateRef.current
   610→    if (ts.moved) return
   611→    const touch = e.changedTouches[0]
   612→    if (!touch) return
   613→    const fakeEvt = { clientX: touch.clientX, clientY: touch.clientY, button: 0, shiftKey: false, preventDefault: () => {} } as any
   614→    handleTap(fakeEvt)
   615→  }
   616→
   617→  // ---- Pointer events (unified mouse + touch) ----
   618→  const handlePointerDown = (e: React.PointerEvent) => {
   619→    if (e.button === 1) {
   620→      mousePanRef.current = {x:e.clientX, y:e.clientY, panning:true}
   621→      return
   622→    }
   623→    if (e.button !== 0 && e.pointerType !== 'touch') return
   624→    const ps = pointerStateRef.current
   625→    ps.startX = e.clientX
   626→    ps.startY = e.clientY
   627→    ps.lastX = e.clientX
   628→    ps.lastY = e.clientY
   629→    ps.moved = false
   630→    ps.isDown = true
   631→  }
   632→
   633→  const handlePointerMove = (e: React.PointerEvent) => {
   634→    const ps = pointerStateRef.current
   635→    if (!ps.isDown) return
   636→
   637→    const dx = e.clientX - ps.lastX
   638→    const dy = e.clientY - ps.lastY
   639→    const totalDx = e.clientX - ps.startX
   640→    const totalDy = e.clientY - ps.startY
   641→
   642→    // If moved > 8px → it's a drag (pan), not a tap
   643→    if (Math.hypot(totalDx, totalDy) > 8) {
   644→      ps.moved = true
   645→      setPan(p => {
   646→        const np = { x: p.x + dx, y: p.y + dy }
   647→        panRef.current = np
   648→        return np
   649→      })
   650→    }
   651→    ps.lastX = e.clientX
   652→    ps.lastY = e.clientY
   653→  }
   654→
   655→  const handlePointerUp = (e: React.PointerEvent) => {
   656→    const ps = pointerStateRef.current
   657→    if (!ps.isDown) return
   658→    ps.isDown = false
   659→
   660→    if (mousePanRef.current) {
   661→      mousePanRef.current.panning = false
   662→    }
   663→
   664→    // If not moved → it's a tap
   665→    if (!ps.moved) {
   666→      const fakeEvt = { clientX: e.clientX, clientY: e.clientY, button: 0, shiftKey: false, preventDefault: () => {} } as any
   667→      handleTap(fakeEvt)
   668→    }
   669→  }
   670→
   671→  // Unified tap/click handler — used by both mouse and touch
   672→  const handleTap = (e: any) => {
   673→    const s = gameRef.current
   674→    const cell = cellFromEvt(e); if (!cell) return
   675→    const pt = pointFromEvt(e)
   676→    const sel = selectedRef.current
   677→    const bm = buildModeRef.current
   678→
   679→    // 1. Build mode — place or cancel
   680→    if (bm) {
   681→      if (placeBuilding(s, 'atreides', bm, cell.x, cell.y)) {
   682→        setBuildMode(null)
   683→      } else {
   684→        setBuildMode(null)
   685→      }
   686→    }
   687→
   688→    // 2. Check what's at tap point
   689→    const tappedUnit = pickUnitAt(s, pt.x, pt.y, 'atreides', 1.5)
   690→    const tappedBld = tappedUnit ? null : pickBuildingAt(s, pt.x, pt.y, 'atreides')
   691→    const tappedEnemy = tappedUnit ? null : pickUnitAt(s, pt.x, pt.y, undefined, 1.5)
   692→    const tappedEnemyBld = (tappedUnit || tappedBld) ? null : pickBuildingAt(s, pt.x, pt.y)
   693→
   694→    // 3. If a BUILDING is selected → tap empty = deselect, tap building = switch
   695→    if (sel?.type === 'building') {
   696→      if (tappedBld) {
   697→        setSelected({ type:'building', id:tappedBld.id })
   698→        setPanelOpen(true)
   699→      }
   700→      if (tappedUnit) {
   701→        setSelected({ type:'unit', id:tappedUnit.id })
   702→      }
   703→      // tap on empty → deselect
   704→      setSelected(null)
   705→      setPanelOpen(false)
   706→    }
   707→
   708→    // 4. If a UNIT is selected
   709→    if (sel?.type === 'unit') {
   710→      const u = s.units.find(u=>u.id===sel.id)
   711→      if (u) {
   712→        // tap on same unit → toggle panel
   713→        if (tappedUnit && tappedUnit.id === u.id) {
   714→          setPanelOpen(p => !p)
   715→        }
   716→        // tap on another friendly unit → switch selection (no panel)
   717→        if (tappedUnit) {
   718→          setSelected({ type:'unit', id:tappedUnit.id })
   719→        }
   720→        // tap on friendly building → switch selection + open panel
   721→        if (tappedBld) {
   722→          setSelected({ type:'building', id:tappedBld.id })
   723→          setPanelOpen(true)
   724→        }
   725→        // tap on enemy → attack order
   726→        if (tappedEnemy && tappedEnemy.owner !== 'atreides') {
   727→          commandAttack(s, u, tappedEnemy.id, false)
   728→          setPanelOpen(false)
   729→        }
   730→        if (tappedEnemyBld && tappedEnemyBld.owner !== 'atreides') {
   731→          commandAttack(s, u, tappedEnemyBld.id, true)
   732→          setPanelOpen(false)
   733→        }
   734→        // tap on empty map → move order
   735→        if (u.type !== 'harvester') {
   736→          commandMove(s, u, cell.x+0.5, cell.y+0.5)
   737→        }
   738→        setPanelOpen(false)
   739→      }
   740→    }
   741→
   742→    // 5. Nothing selected
   743→    if (tappedBld) {
   744→      // building → select + open panel immediately
   745→      setSelected({ type:'building', id:tappedBld.id })
   746→      setPanelOpen(true)
   747→    } else if (tappedUnit) {
   748→      // unit → select (no panel — opens on 2nd tap)
   749→      setSelected({ type:'unit', id:tappedUnit.id })
   750→    } else {
   751→      // empty → nothing
   752→    }
   753→  }
   754→
   755→  // Deselect function (called by ✕ button in HUD or Esc)
   756→  const deselect = () => {
   757→    setSelected(null)
   758→    setPanelOpen(false)
   759→    setBuildMode(null)
   760→  }
   761→  // reset zoom/pan on Esc
   762→  useEffect(() => {
   763→    const k = (e: KeyboardEvent) => { if (e.key === 'Escape') { setBuildMode(null); setSelected(null); setPanelOpen(false); setZoom(1.0); centerOnBase() } }
   764→    window.addEventListener('keydown', k)
   765→    return () => window.removeEventListener('keydown', k)
   766→  }, [])
   767→
   768→  // ---- game loop (logic) ----
   769→  // Tick at 10fps; throttle React re-renders to ~4fps (UI panel doesn't need 10fps)
   770→  useEffect(() => {
   771→    let tickCount = 0
